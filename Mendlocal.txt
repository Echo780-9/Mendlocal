<script>
let requests = JSON.parse(localStorage.getItem('mendlocal_requests')) || [];
let karma = parseInt(localStorage.getItem('mendlocal_karma')) || 5;

function saveRequests(){
    localStorage.setItem('mendlocal_requests', JSON.stringify(requests));
    renderRequests();
}

function saveKarma(){
    localStorage.setItem('mendlocal_karma', karma.toString());
    document.getElementById('karma-score').textContent = karma;
}

function showPage(pageId){
    ['home','post','karma','about'].forEach(p=>{
        document.getElementById('page-'+p).classList.add('hidden');
        document.querySelector(`.nav-btn:nth-child(${['home','post','karma','about'].indexOf(p)+1})`).classList.remove('active');
    });
    document.getElementById('page-'+pageId).classList.remove('hidden');
    document.querySelector(`.nav-btn:nth-child(${['home','post','karma','about'].indexOf(pageId)+1})`).classList.add('active');
    if(pageId==='karma') saveKarma();
}

function postRequest(){
    const title = document.getElementById('new-title').value.trim();
    const desc = document.getElementById('new-desc').value.trim();
    const category = document.getElementById('new-category').value;
    const distance = document.getElementById('new-distance').value;
    const urgent = document.getElementById('new-urgent').checked;

    if(!title){alert('Please describe what you need help with!');return;}

    requests.unshift({
        id: Date.now(),
        title,
        desc,
        category,
        distance,
        urgent,
        status: 'open',
        time: new Date().toLocaleString()
    });

    document.getElementById('new-title').value = '';
    document.getElementById('new-desc').value = '';
    saveRequests();
    showPage('home');
    alert('✅ Posted! Your request is now visible to your local community. Thank you for being part of MendLocal 💚');
}

function updateStatus(id, newStatus){
    const req = requests.find(r=>r.id===id);
    if(req){
        req.status = newStatus;
        if(newStatus==='done') karma += 1;
        saveRequests();
        saveKarma();
    }
}

function deleteRequest(id){
    if(confirm('Remove this request?')){
        requests = requests.filter(r=>r.id!==id);
        saveRequests();
    }
}

function filterRequests(category){
    document.querySelectorAll('.tab').forEach(t=>t.classList.remove('active'));
    event.target.classList.add('active');
    renderRequests(category);
}

function renderRequests(filter='all'){
    const list = document.getElementById('requests-list');
    const filtered = filter==='all' ? requests : 
                      filter==='urgent' ? requests.filter(r=>r.urgent) : 
                      requests.filter(r=>r.category===filter);

    document.getElementById('empty-message').style.display = filtered.length ? 'none' : 'block';

    list.innerHTML = filtered.map(req => `
        <div class="card">
            <div style="display:flex;justify-content:space-between;align-items:flex-start;margin-bottom:8px;">
                <div>
                    ${req.urgent ? '<span class="tag tag-urgent">🚨 URGENT</span>' : ''}
                    <span class="tag tag-${req.category}">${req.category.charAt(0).toUpperCase()+req.category.slice(1)}</span>
                </div>
                <span class="request-status status-${req.status}">
                    ${req.status==='open'?'🟢 Open':req.status==='taken'?'🟡 In Progress':'✅ Done'}
                </span>
            </div>
            <h4 style="margin:8px 0 4px;font-size:17px;">${req.title}</h4>
            <p style="font-size:14px;color:var(--grey);margin-bottom:8px;">${req.desc}</p>
            <p class="distance">📍 ${req.distance} · Posted ${req.time}</p>
            
            ${req.status==='open' ? `
                <div style="display:flex;gap:8px;margin-top:10px;">
                    <button class="btn btn-green btn-small" onclick="updateStatus(${req.id},'taken')">🤝 I Can Help!</button>
                    <button class="btn btn-outline btn-small" onclick="updateStatus(${req.id},'done')">✅ Mark Done</button>
                    <button class="btn btn-small" style="background:#f1f1f1;color:#666;border:none;" onclick="deleteRequest(${req.id})">🗑️</button>
                </div>
            ` : req.status==='taken' ? `
                <div style="display:flex;gap:8px;margin-top:10px;">
                    <button class="btn btn-green btn-small" onclick="updateStatus(${req.id},'done')">✅ Mark Done</button>
                    <button class="btn btn-outline btn-small" onclick="updateStatus(${req.id},'open')">↩️ Still Needs Help</button>
                </div>
            ` : `
                <p style="color:var(--green);margin-top:10px;">✅ Fixed! Thanks to our neighbour who helped 💚</p>
                <button class="btn btn-small" style="margin-top:6px;background:#f1f1f1;color:#666;border:none;" onclick="deleteRequest(${req.id})">Remove</button>
            `}
        </div>
    `).join('');
}

renderRequests();
saveKarma();
</script>
